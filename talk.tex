% Risc-v на сегодняшний день является одной из наиболее активно развивающихся открытых архитектур.

% Важную роль в экосистеме risc-v играет проект LLVM. LLVM --- инфраструктура для создания компиляторов. LLVM имеет множество целевых плафторм, в том числе и RISC-V. Спецификация ISA целевой архитектуры описывается в LLVM декларативно, на специальном декларативном языке, который называется TableGen.

% Одноврменно, существует язык Sail, предназначенный для определения моделей процессорных архитектур. Существует формальная

% Одноврменно, существует язык Sail, предназначенный для определения моделей процессорных архитектур. На Sail существует оффициальная, одобренная сообществом risc-v foundation модель risc-v isa. Модель представляет из себя поведенческую спецификацию.

% Несмотря на наличие двух независимых спецификаций RISC-V ISA, в настоящее время не существует инструментов, позволяющих систематически сопоставлять эти спецификации и проверять их согласованность. Это создаёт риск накопления расхождений по мере их эволюции.
% Наличие расхождений может приводить к некорректной генерации кода компилятором и снижению общей надёжности экосистемы RISC-V. Поэтому актуальна проблема валидации спецификаций относительно друг друга

\begin{frame}{Введение}
    \begin{itemize}
        \item RISC-V --- открытая, расширяемая ISA (Instruction Set Architecture)

        \item Sail\footnote[frame]{\url{https://github.com/rems-project/sail/}} --- язык для описания моделей процессорных архитектур. Позволяет задать семантику исполнения процессорных инструкций
              \begin{itemize}
                  \item Sail-riscv\footnote[frame]{\url{https://github.com/riscv/sail-riscv}} --- формальная, поведенческая спецификация RISC-V ISA
              \end{itemize}

        \item LLVM\footnote[frame]{\url{https://llvm.org}} --- инфраструктура для компиляторов
              \begin{itemize}
                  \item Спецификация целевой ISA описывается декларативно на TableGen
              \end{itemize}

        \item Актуальна проблема валидации спецификаций относительно друг друга
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Обзор. Описание спецификации в LLVM}

    \begin{itemize}
        \item TableGen\footnote[frame]{\url{https://llvm.org/docs/TableGen/}} --- декларативный язык, который используется в LLVM для описания предметно-ориентированной информации
        \item В частности, для описания спецификации ISA целевых платформ
        \item Для каждой инструкции описываются следующие свойства:

                  \item Может ли \textbf{читать из памяти}, \textbf{писать в память}
                  \item Коммутативность
                  \item ...
              \end{itemize}
        \item Файлы с TableGen кодом имеют расширение \texttt{.td}
        \item При помощи утилиты \texttt{llvm-tblgen} можно скомпилировать в \texttt{JSON}
    \end{itemize}

\begin{frame}[fragile]{Обзор. Sail-riscv}


    \begin{itemize}
        \item Sail-riscv\footnote[frame]{\url{https://github.com/riscv/sail-riscv}} --- это модель архитектуры RISC-V на Sail
        \item Модель представляет из себя реализацию интерпретатора инструкций RISC-V ISA
              % \item Модель можно скомпилировать в эмулятор на \C{} или \OCaml{}, который интерпретирует \texttt{elf} файлы

              \begin{itemize}
                  \item \texttt{AST}
                  \item \texttt{encdec}, кодирование/декодирование кодов инструкций (opcode) в AST
                  \item \texttt{assembly}, отображение AST в мнемоники
                  \item \texttt{execute}, семантика исполнения инструкций
              \end{itemize}

    \end{itemize}


    \begin{lstlisting}[caption={}, language=sail, frame=single]
function clause execute( ADD(rs2, rs1, rd) ) = {
    let rs1_val = X(rs1);                  // read the rs1
    let rs2_val = X(rs2);                  // read the rs2
    let result : xlenbits = rs1_val + rs2_val; // compute
    X(rd) = result;                        // write result to rd
    RETIRE_SUCCESS
}
    \end{lstlisting}

\end{frame}

\begin{frame}[fragile]{Реализация. Статический анализатор}

    \begin{itemize}
        \item Язык реализации --- \OCaml{}

        \item \textbf{Sail}
              \begin{enumerate}
                  \item Входные данные --- \texttt{sail} файлы модели
                  \item Синтаксический анализ файлов при помощи библиотеки компилятора Sail
                  \item Статический анализ
                        \begin{itemize}
                            \item Определение мнемоник
                            \item Специализация \texttt{execute} функций
                            \item Анализ \texttt{execute} функций
                        \end{itemize}
              \end{enumerate}

        \item \textbf{LLVM TableGen}
              \begin{enumerate}
                  \item Входные данные --- файлы спецификации на TableGen, скомпилированные\footnote[frame]{при помощи \texttt{llvm-tblgen}} в \texttt{JSON}
                  \item Из \texttt{JSON} извлекается
                        \begin{itemize}
                            \item Входные и выходные операнды
                            \item Может ли инструкция читать/писать память
                        \end{itemize}
              \end{enumerate}

        \item Полученная информация сравнивается, и генерируется отчёт, содержащий расхождения
    \end{itemize}

\end{frame}

\begin{frame}[fragile]
    \frametitle{Дополнительный слайд. Sail-riscv (3)}

    \begin{lstlisting}[caption={Пример определения семантики исполнения инструкций \texttt{addi}, \texttt{slti}, \texttt{sltiu}, \texttt{andi}, \texttt{ori} и \texttt{xori}}, language=sail, frame=single]
function clause execute (ITYPE (imm, rs1, rd, op)) = {
  let rs1_val = X(rs1);               // read the rs1
  let immext : xlenbits = EXTS(imm);  // sign-extend imm
  let result : xlenbits = match op {  // compute the result
    RISCV_ADDI  => rs1_val + immext,  // for addi, do addition
    RISCV_SLTI  => EXTZ(rs1_val <_s immext), // ...etc
    ...
  };
  X(rd) = result;                     // write result to rd
  RETIRE_SUCCESS                      // successful termination
}
\end{lstlisting}

\end{frame}
